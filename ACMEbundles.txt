// Initialize a new bundle
void add_bundle(const char *name, const char *board, bool is_default, double transfer_time, double release_time, bool wait_for_data) {
    if (bundle_count >= MAX_BUNDLES) {
        printf("Error: Maximum number of bundles reached.\n");
        return;
    }
    strcpy(bundles[bundle_count].bundle_name, name);
    strcpy(bundles[bundle_count].io_board, board);
    bundles[bundle_count].is_default_bundle = is_default;
    bundles[bundle_count].transfer_time = transfer_time;
    bundles[bundle_count].release_time = release_time;
    bundles[bundle_count].wait_for_data = wait_for_data;
    bundles[bundle_count].channel_count = 0;
    bundle_count++;
}

// Add a channel to a bundle
void add_channel_to_bundle(const char *bundle_name, int channel) {
    for (int i = 0; i < bundle_count; i++) {
        if (strcmp(bundles[i].bundle_name, bundle_name) == 0) {
            if (bundles[i].channel_count >= MAX_CHANNELS) {
                printf("Error: Maximum channels for bundle %s reached.\n", bundle_name);
                return;
            }
            bundles[i].channels[bundles[i].channel_count++] = channel;
            return;
        }
    }
    printf("Error: Bundle %s not found.\n", bundle_name);
}

// Simulate data transfer for a bundle
void simulate_transfer(double current_time, const ChannelBundle *bundle) {

    if (bundle->wait_for_data) {

    } else {
        printf("-> Assuming data is fresh. Processing without wait.\n");
    }
    printf("Transfer Complete for Bundle: %s\n", bundle->bundle_name);
}
