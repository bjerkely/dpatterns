class TLECommand:
    """ Base class for all TLE commands """
    def __init__(self, cmd_type):
        self.cmd_type = cmd_type

    def execute(self, tle):
        pass


class CMD_JUMP(TLECommand):
    """ Command to jump if the trigger ID matches """
    def __init__(self, trig_id, jump_addr):
        super().__init__(cmd_type="CMD_JUMP")
        self.trig_id = trig_id
        self.jump_addr = jump_addr

    def execute(self, tle):
        if tle.trigger_id == self.trig_id:
            tle.jump(self.jump_addr)
        else:
            tle.next_command()

class TLE:
    """ Transaction List Engine class """
    def __init__(self, trigger_id=0):
        self.trigger_id = trigger_id
        self.command_list = []
        self.pc = 0  # Program Counter

    def load_command_list(self, commands):
        self.command_list = commands

    def execute_commands(self):
        while self.pc < len(self.command_list):
            command = self.command_list[self.pc]
            command.execute(self)

    def next_command(self):
        self.pc += 1

    def jump(self, addr):
        self.pc = addr

    def transfer(self, loc_addr, dest_addr, length, dest_id):
        print(f"Transferring {length} bytes from {loc_addr} to {dest_addr} for destination {dest_id}")

    def notify(self, info_value, dest_id):
        print(f"Notification sent with value {info_value} to destination {dest_id}")

    def postpone(self, cnt):
        print(f"Postponing execution by {cnt * 12.5} ns")

    def stop_execution(self):
        print("Execution stopped.")
        self.pc = len(self.command_list)  # Terminate the loop

1. Introduction
The Transaction List Engine (TLE) is a Direct Memory Access (DMA)-based system responsible for transferring blocks of data from the application memory space or SDD registers to different endpoints in the Serial RapidIO (SRIO) fabric. The TLE is initialized and controlled by a series of commands in a Transaction Command List (TCL), which is executed when specific Trigger IDs are received.

The system supports efficient data transfer between different clients or devices in an SRIO fabric, ensuring controlled data flow, error handling, and flexible execution management through various command types.

2. Components of the TLE System
2.1 Transaction Command List (TCL)
The TCL is a list of commands executed by the TLE when a specific Trigger ID is received. The TCL defines the sequence of operations, such as data transfers, jumps to other parts of the list, or halting execution. The commands in the TCL are written and initialized by software before the TLE is enabled.

The command list memory is allocated based on application needs, and the sequence can contain a combination of transfer, jump, and stop commands.
